# Este archivo contiene las consultas SQL necesarias para demostrar
# el funcionamiento completo del proyecto BD2 con sus 4 tipos de índices.
#
# TIPOS DE ÍNDICES IMPLEMENTADOS:
# 1. B+ Tree - Búsqueda eficiente por clave primaria
# 2. Extendible Hash - Acceso directo por clave con manejo de colisiones
# 3. Sequential File - Acceso secuencial optimizado
# 4. R-Tree - Índice espacial para consultas geográficas

# ================================================================
# 1. DEMOSTRACIÓN B+ TREE
# ================================================================
# Búsqueda por clave primaria usando índice B+ Tree
# Muestra acceso eficiente O(log n) con estadísticas de I/O

-- Búsqueda puntual en tabla B+ Tree (1K registros)
SELECT * FROM books_bpt_1k WHERE book_key = 1;

-- Búsqueda puntual en tabla B+ Tree (diferente clave)  
SELECT * FROM books_bpt_1k WHERE book_key = 500;

-- Búsqueda por rango usando BETWEEN
SELECT * FROM books_bpt_1k WHERE book_key BETWEEN 100 AND 200;

# ================================================================
# 2. DEMOSTRACIÓN EXTENDIBLE HASH
# ================================================================
# Búsqueda por clave usando índice Hash Extensible
# Muestra acceso directo O(1) con manejo de colisiones

-- Búsqueda puntual en tabla Hash (1K registros)
SELECT * FROM books_hash_1k WHERE book_key = 1;

-- Búsqueda puntual con clave diferente
SELECT * FROM books_hash_1k WHERE book_key = 999;

-- Búsqueda con clave que probablemente cause colisión
SELECT * FROM books_hash_1k WHERE book_key = 256;

# ================================================================
# 3. DEMOSTRACIÓN SEQUENTIAL FILE
# ================================================================
# Búsqueda usando acceso secuencial optimizado
# Muestra escaneo eficiente con estadísticas de I/O

-- Búsqueda puntual en tabla Sequential (1K registros)
SELECT * FROM books_seq_1k WHERE book_key = 1;

-- Búsqueda por rango en archivo secuencial
SELECT * FROM books_seq_1k WHERE book_key BETWEEN 100 AND 200;

-- Búsqueda con LIMIT para mostrar eficiencia
SELECT * FROM books_seq_1k WHERE book_key > 500 LIMIT 10;

# ================================================================
# 4. DEMOSTRACIÓN R-TREE (ÍNDICES ESPACIALES)
# ================================================================
# Consultas geográficas con visualización en plano
# El frontend mostrará automáticamente la visualización espacial

-- Búsqueda por radio: encuentra todos los puntos dentro de un círculo
-- Coordenadas: (29.266858, 41.047655) - Estambul, Turquía
-- Radio: 100 unidades
-- RESULTADO: Visualización con círculo de búsqueda y puntos encontrados
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(29.266858, 41.047655), RADIUS 100);

-- Búsqueda KNN: encuentra los 5 puntos más cercanos
-- Coordenadas: (29.266858, 41.047655) - Estambul, Turquía
-- K: 5 vecinos más cercanos
-- RESULTADO: Visualización con líneas de conexión a los 5 vecinos más cercanos
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(29.266858, 41.047655), K 5);

-- Búsqueda por radio en Nueva York
-- Coordenadas: (-74.0060, 40.7128) - Nueva York, EE.UU.
-- Radio: 50 unidades
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(-74.0060, 40.7128), RADIUS 50);

-- Búsqueda KNN en Londres
-- Coordenadas: (-0.1278, 51.5074) - Londres, Reino Unido
-- K: 10 vecinos más cercanos
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(-0.1278, 51.5074), K 10);

# ================================================================
# 7. NOTAS PARA LA PRESENTACIÓN
# ================================================================
#
# CARACTERÍSTICAS A DEMOSTRAR:
#
# 1. B+ TREE:
#    - Acceso eficiente O(log n)
#    - Buen rendimiento para búsquedas por rango
#    - Balance automático del árbol
#
# 2. EXTENDIBLE HASH:
#    - Acceso directo O(1) en caso ideal
#    - Manejo dinámico de colisiones
#    - Rehashing automático cuando es necesario
#
# 3. SEQUENTIAL FILE:
#    - Acceso secuencial optimizado
#    - Buen rendimiento para consultas de rango
#    - Simplicidad y fiabilidad
#
# 4. R-TREE:
#    - Índice espacial para coordenadas geográficas
#    - Búsqueda por radio (circular)
#    - Búsqueda KNN (k vecinos más cercanos)
#    - Visualización automática en el frontend
#
# MÉTRICAS A OBSERVAR:
# - Tiempo de ejecución (ms)
# - Operaciones de I/O (lecturas/escrituras)
# - Uso de memoria
# - Escalabilidad con diferentes volúmenes de datos
