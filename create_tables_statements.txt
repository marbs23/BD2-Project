# 0. CREACIÓN DE TABLAS PARA DEMOSTRACIÓN
# ================================================================
# Estas sentencias CREATE TABLE deben ejecutarse ANTES de las consultas
# de demostración para asegurar que todas las tablas existan

-- Crear tabla B+ Tree para libros (1K registros)
CREATE TABLE books_bpt_1k (
    book_key INT INDEX BPTREE,
    title TEXT,
    author TEXT,
    pages INT,
    rating FLOAT,
    year INT
) FROM FILE "./books/books_1000.csv";

-- Crear tabla Hash para libros (1K registros)
CREATE TABLE books_hash_1k (
    book_key INT INDEX HASH,
    title TEXT,
    author TEXT,
    pages INT,
    rating FLOAT,
    year INT
) FROM FILE "./books/books_1000.csv";

-- Crear tabla Sequential para libros (1K registros)
CREATE TABLE books_seq_1k (
    book_key INT,
    title TEXT,
    author TEXT,
    pages INT,
    rating FLOAT,
    year INT
) FROM FILE "./books/books_1000.csv";

-- Crear tabla R-Tree para coordenadas (10K registros)
CREATE TABLE coords_rtree_10k_rtree (
    point_id INT INDEX RTREE,
    lon FLOAT,
    lat FLOAT
) FROM FILE "./coords/coords_10000.csv";

-- NOTA: Las tablas deben crearse en este orden para evitar conflictos
-- Las tablas B+ Tree y Sequential usan BPTREE por defecto en el sistema
-- La tabla R-Tree requiere la columna con índice RTREE para consultas espaciales

# ================================================================
# 1. DEMOSTRACIÓN B+ TREE
# ================================================================
# Búsqueda por clave primaria usando índice B+ Tree
# Muestra acceso eficiente O(log n) con estadísticas de I/O

-- Búsqueda puntual en tabla B+ Tree (1K registros)
SELECT * FROM books_bpt_1k WHERE book_key = 2145253;

-- Búsqueda puntual en tabla B+ Tree (diferente clave)  
SELECT * FROM books_bpt_1k WHERE book_key = 958696;

-- Búsqueda por rango usando BETWEEN
SELECT * FROM books_bpt_1k WHERE book_key BETWEEN 1000000 AND 2000000;

# ================================================================
# 2. DEMOSTRACIÓN EXTENDIBLE HASH
# ================================================================
# Búsqueda por clave usando índice Hash Extensible
# Muestra acceso directo O(1) con manejo de colisiones

-- Búsqueda puntual en tabla Hash (1K registros)
SELECT * FROM books_hash_1k WHERE book_key = 2145253;

-- Búsqueda puntual con clave diferente
SELECT * FROM books_hash_1k WHERE book_key = 1793387;

-- Búsqueda con clave que probablemente cause colisión
SELECT * FROM books_hash_1k WHERE book_key = 916302;

# ================================================================
# 3. DEMOSTRACIÓN SEQUENTIAL FILE
# ================================================================
# Búsqueda usando acceso secuencial optimizado
# Muestra escaneo eficiente con estadísticas de I/O

-- Búsqueda puntual en tabla Sequential (1K registros)
SELECT * FROM books_seq_1k WHERE book_key = 2145253;

-- Búsqueda por rango en archivo secuencial
SELECT * FROM books_seq_1k WHERE book_key BETWEEN 1000000 AND 2000000;

-- Búsqueda con LIMIT para mostrar eficiencia
SELECT * FROM books_seq_1k WHERE book_key > 2000000 LIMIT 10;

# ================================================================
# 4. DEMOSTRACIÓN R-TREE (ÍNDICES ESPACIALES)
# ================================================================
# Consultas geográficas con visualización en plano
# El frontend mostrará automáticamente la visualización espacial

-- Búsqueda por radio: encuentra todos los puntos dentro de un círculo
-- Coordenadas: (-47.074277, -24.045516) - Primer punto del dataset
-- Radio: 100 unidades
-- RESULTADO: Visualización con círculo de búsqueda y puntos encontrados
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(-47.074277, -24.045516), RADIUS 100);

-- Búsqueda KNN: encuentra los 5 puntos más cercanos
-- Coordenadas: (2.868954, 49.109141) - Segundo punto del dataset
-- K: 5 vecinos más cercanos
-- RESULTADO: Visualización con líneas de conexión a los 5 vecinos más cercanos
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(2.868954, 49.109141), K 5);

-- Búsqueda por radio en Nueva York
-- Coordenadas: (-74.0060, 40.7128) - Nueva York, EE.UU.
-- Radio: 50 unidades
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(-74.0060, 40.7128), RADIUS 50);

-- Búsqueda KNN en Londres
-- Coordenadas: (-0.1278, 51.5074) - Londres, Reino Unido
-- K: 10 vecinos más cercanos
SELECT * FROM coords_rtree_10k_rtree WHERE point_id IN (POINT(-0.1278, 51.5074), K 10);

# 7. NOTAS PARA LA PRESENTACIÓN

#
# CARACTERÍSTICAS A DEMOSTRAR:
#
# 1. B+ TREE:
#    - Acceso eficiente O(log n)
