-- Sentencias SQL para crear tablas con diferentes índices y tamaños
-- Formato: CREATE TABLE nombre_tabla (columnas) FROM FILE "ruta_archivo";

-- Tablas con índice BPTREE (soportan SELECT * sin WHERE)
CREATE TABLE books_bpt_1k (
book_key INT INDEX BPTREE,
title TEXT,
author TEXT,
pages FLOAT,
average_rating FLOAT,
published_date FLOAT
) FROM FILE "books/books_1000.csv";

CREATE TABLE books_bpt_10k (
book_key INT INDEX BPTREE,
title TEXT,
author TEXT,
pages FLOAT,
average_rating FLOAT,
published_date FLOAT
) FROM FILE "books/books_10000.csv";

CREATE TABLE books_bpt_100k (
book_key INT INDEX BPTREE,
title TEXT,
author TEXT,
pages FLOAT,
average_rating FLOAT,
published_date FLOAT
) FROM FILE "books/books_100000.csv";

CREATE TABLE books_bpt_1m (
book_key INT INDEX BPTREE,
title TEXT,
author TEXT,
pages FLOAT,
average_rating FLOAT,
published_date FLOAT
) FROM FILE "books/books_1000000.csv";

-- Tablas con índice HASH (solo soportan SELECT con WHERE exacto)
CREATE TABLE books_hash_1k (
book_key INT INDEX HASH,

) FROM FILE "books/books_1000.csv";

CREATE TABLE books_hash_10k (

CREATE TABLE books_hash_100k (

) FROM FILE "books/books_100000.csv";

-- Tablas con índice SEQUENTIAL (soportan SELECT * sin WHERE)
CREATE TABLE books_seq_1k (
book_key INT INDEX SEQUENTIAL,
title TEXT,

pages FLOAT,
average_rating FLOAT,
published_date FLOAT
) FROM FILE "books/books_1000.csv";

CREATE TABLE books_seq_10k (

) FROM FILE "books/books_10000.csv";

CREATE TABLE books_seq_100k (

CREATE TABLE books_seq_1m (

) FROM FILE "books/books_1000000.csv";

-- Tablas con índice RTREE (requieren columnas de coordenadas lon/lat)
CREATE TABLE coords_rtree_1k (
point_id INT INDEX RTREE,
lon FLOAT,
lat FLOAT
) FROM FILE "coords/coords_1000.csv";

CREATE TABLE coords_rtree_10k (
point_id INT INDEX RTREE,
lon FLOAT,

) FROM FILE "coords/coords_10000.csv";

CREATE TABLE coords_rtree_100k (
point_id INT INDEX RTREE,

) FROM FILE "coords/coords_100000.csv";

CREATE TABLE coords_rtree_1m (
point_id INT INDEX RTREE,

) FROM FILE "coords/coords_1000000.csv";
